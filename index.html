Mã Nguồn Mẫu Nạp Vào ESP8266 (Arduino C++)

Đây là mã Arduino C++ chuẩn sử dụng thư viện PubSubClient và DHT sensor library để gửi gói tin JSON đúng mẫu {"temperature":"..", "humidity":".."} lên EMQX qua cổng TCP MQTTS 8883.

ESP8266_EMQX_DHT11.ino 
#include <ESP8266WiFi.h>
#include <WiFiClientSecure.h>
#include <PubSubClient.h>
#include <DHT.h>

// --- CẤU HÌNH WIFI ---
const char* ssid = "TEN_WIFI_CUA_BAN";
const char* password = "MAT_KHAU_WIFI";

// --- CẤU HÌNH EMQX BROKER ---
const char* mqtt_server = "wd817fa2.ala.asia-southeast1.emqxsl.com";
const int mqtt_port = 8883;             // Cổng MQTTS TCP trên ESP8266
const char* mqtt_user = "iot";          // MQTT username
const char* mqtt_pass = "P557724";      // MQTT password
const char* mqtt_topic = "esp8266/dht11";

// --- CẤU HÌNH DHT11 ---
#define DHTPIN D4        // Chân Data của DHT11 nối chân D4 (GPIO2)
#define DHTTYPE DHT11
DHT dht(DHTPIN, DHTTYPE);

WiFiClientSecure espClient;
PubSubClient client(espClient);

unsigned long lastMsg = 0;

void setup_wifi() {
  delay(10);
  Serial.print("Connecting to WiFi: ");
  Serial.println(ssid);
  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  Serial.println("\nWiFi connected! IP: " + WiFi.localIP().toString());
  
  // Bỏ qua kiểm tra chứng chỉ SSL (thích hợp cho ESP8266)
  espClient.setInsecure();
}

void reconnect() {
  while (!client.connected()) {
    Serial.print("Attempting MQTT connection to EMQX...");
    String clientId = "ESP8266Client-" + String(random(0xffff), HEX);
    if (client.connect(clientId.c_str(), mqtt_user, mqtt_pass)) {
      Serial.println("CONNECTED!");
    } else {
      Serial.print("failed, rc=");
      Serial.print(client.state());
      Serial.println(" try again in 5 seconds");
      delay(5000);
    }
  }
}

void setup() {
  Serial.begin(115200);
  dht.begin();
  setup_wifi();
  client.setServer(mqtt_server, mqtt_port);
}

void loop() {
  if (!client.connected()) {
    reconnect();
  }
  client.loop();

  // Gửi số liệu mỗi 5 giây
  unsigned long now = millis();
  if (now - lastMsg > 5000) {
    lastMsg = now;

    float h = dht.readHumidity();
    float t = dht.readTemperature();

    if (isnan(h) || isnan(t)) {
      Serial.println("Lỗi đọc cảm biến DHT11!");
      return;
    }

    // Đóng gói dạng JSON đúng yêu cầu:
    // {"temperature":"78.00", "humidity":"78.00"}
    char jsonBuffer[100];
    snprintf(jsonBuffer, sizeof(jsonBuffer), 
             "{\"temperature\":\"%.2f\", \"humidity\":\"%.2f\"}", t, h);

    Serial.print("Publishing message: ");
    Serial.println(jsonBuffer);
    client.publish(mqtt_topic, jsonBuffer);
  }
}
